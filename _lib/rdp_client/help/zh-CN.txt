# 请先创建专用入口命令，例如：copy Favorites\template.rdp1.cmd myrdp.cmd，然后修改 myrdp.cmd 中的绑定信息

# 基础用法示例：
  {{COMMAND}}                               启动远程桌面（会生成或复用已生成的 .rdp 文件）
  {{COMMAND}} .help en                      Show English help.
  {{COMMAND}} .help zh                      显示中文帮助
  {{COMMAND}} .list                         列出对端会话ID；依赖 SSH 远程-即入口命令中须指定 RDP_PEER_SSH_ENTRY
  {{COMMAND}} .2                            以普通 RDP 精确连接会话 ID 2；必要时自动切换并验证
  {{COMMAND}} .2 screenshot                 截取会话 2，不改变其显示路由
  {{COMMAND}} .2 screenshot --display       必要时临时建立可渲染的 RDP 显示，截图完成后关闭
  {{COMMAND}} .2 screenshot --output D:\shot.png 保存到绝对 PNG 路径；目标文件必须尚不存在
  {{COMMAND}} .2 pixel 640 360              按会话截图坐标读取像素
  {{COMMAND}} .2 click 640 360              按会话截图坐标左键点击；会话未就绪时追加 --display
  {{COMMAND}} .2 screenshot --display --timeout 60s 所有桌面工作阶段共享同一份 60 秒预算
  {{COMMAND}} .rdp create                   仅生成 RDP 文件；目标已存在时拒绝覆盖
  {{COMMAND}} .rdp create --force           生成 RDP 文件并覆盖已有目标

  桌面动作依赖 RDP_PEER_SSH_ENTRY 和托管 PsExec；请先运行 {{COMMAND}} .peer psexec add。
  活动但锁定的会话会直接拒绝桌面动作；--display 不会自动解锁或改路由。


# 管理对端开启或关闭 Shadow 模式（依赖 SSH 远程-即入口命令中须指定 RDP_PEER_SSH_ENTRY）：
  {{COMMAND}} .peer shadow status               查看对端 Shadow 状态及本工具可恢复的改动
  {{COMMAND}} .peer shadow enable               启用跨主机 Shadow 的对端网络条件
  {{COMMAND}} .peer shadow enable --dry-run     查看 enable 将进行的修改
  {{COMMAND}} .peer shadow mode <0-4>           设置对端整机 Shadow 模式：
                                         0=禁止；1=经同意控制；2=免同意控制；3=经同意查看；4=免同意查看
  {{COMMAND}} .peer shadow mode <0-4> --dry-run 查看 mode 将进行的修改
  {{COMMAND}} .peer shadow restore              恢复本工具修改过的对端 Shadow 设置
  {{COMMAND}} .peer shadow restore --dry-run    查看 restore 将进行的修改



# Shadow 模式：
  {{COMMAND}} .shadow doctor                         检查当前客户端到对端的 Shadow 前置条件
  {{COMMAND}} .shadow 2                              [交互]申请查看 ID 为 2 的会话（对端会弹出，是否同意的对话框）
  {{COMMAND}} .shadow 2 --no-consent                 查看 ID 为 2 的会话，免目标用户同意（要控制请追加 --control）
  {{COMMAND}} .shadow 3 --control --no-consent       控制 ID 为 3 的会话；不加 --control 的均为查看
  {{COMMAND}} .shadow console --no-consent           查看当前活动的 console 会话（依赖 SSH 查询具体会话 ID）
  {{COMMAND}} .shadow console --tscon 3 --no-consent 明确将会话 3 挂接到 console 后查看；可能断开原 console 会话
  {{COMMAND}} .shadow console --display --no-consent console 无活动桌面时，以临时普通 RDP 建立桌面并安全挂接后查看


# 通过 SSH 管理和调用对端 PsExec：
  {{COMMAND}} .peer psexec status           查看对端机 PsExec 可用情况
  {{COMMAND}} .peer psexec add              下载 PsExec 拷贝到对端机指定位置，待用
  {{COMMAND}} .peer psexec add --dry-run    同上，但仅预览要做的操作，不真实执行
  {{COMMAND}} .peer psexec remove           add 的反向操作
  {{COMMAND}} .peer psexec remove --dry-run 同上，但仅预览要做的操作，不真实执行
  {{COMMAND}} .peer psexec 2 notepad.exe    在会话 2 的桌面，打开记事本（窗口程序）
  {{COMMAND}} .peer psexec -- <原生参数...> 在对端机上执行原生 PsExec 命令（会自动补 -accepteula）


# 通过系统 hosts 管理别名（实现按账户区分凭据）：
  {{COMMAND}} .hosts status                 查看本入口别名的 hosts 托管状态
  {{COMMAND}} .hosts install                非交互安装或修复；权限不足时直接报错
  {{COMMAND}} .hosts install --uac          安装或修复托管映射；仅在需要提权时弹出 UAC
  {{COMMAND}} .hosts remove                 非交互删除本入口映射；权限不足时直接报错
  {{COMMAND}} .hosts remove --uac           删除本入口创建的映射；仅在需要提权时弹出 UAC
  {{COMMAND}} .hosts cleanup --dry-run      显示可清理的托管映射，不修改 hosts
  {{COMMAND}} .hosts cleanup                非交互清理失去入口的托管映射；权限不足时直接报错
  {{COMMAND}} .hosts cleanup --uac          清理失去入口的托管映射；仅在需要提权时弹出 UAC

# .rdp 文件自签名管理：
  {{COMMAND}} .sign status                  查看 My、Root 和 mstsc 信任策略状态
  {{COMMAND}} .sign install                 创建 My 私钥证书、Root 公钥副本和 mstsc 信任项
  {{COMMAND}} .sign install --dry-run       列出上述 ADD，不修改证书库或注册表
  {{COMMAND}} .sign remove                  删除上述专用证书及其 mstsc 信任项
  {{COMMAND}} .sign open                    打开“证书 - 当前用户”管理器
